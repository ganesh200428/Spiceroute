ORDERS = {}

def get_cart(): return session.get("cart", {})
def save_cart(cart): session["cart"] = cart
def get_restaurant(rid): return next((r for r in RESTAURANTS if r["id"]==rid), None)
def get_item(iid):
    for items in MENUS.values():
        for item in items:
            if item["id"] == iid: return item
    return None

@app.route("/api/restaurants")
def api_restaurants(): return jsonify(RESTAURANTS)

@app.route("/api/menu/<rid>")
def api_menu(rid):
    r = get_restaurant(rid)
    if not r: return jsonify({"error":"Not found"}), 404
    menu = MENUS.get(rid, [])
    cats = {}
    for item in menu: cats.setdefault(item["category"], []).append(item)
    return jsonify({"restaurant": r, "menu": cats})

@app.route("/api/cart")
def api_cart_get():
    cart = get_cart()
    items, total = [], 0
    for iid, qty in cart.items():
        item = get_item(iid)
        if item:
            sub = item["price"] * qty
            total += sub
            items.append({**item, "quantity": qty, "subtotal": sub})
    return jsonify({"items": items, "total": total, "count": sum(cart.values())})

@app.route("/api/cart/add", methods=["POST"])
def api_cart_add():
    data = request.get_json()
    iid = data.get("item_id")
    if not get_item(iid): return jsonify({"error":"Not found"}), 404
    cart = get_cart()
    cart[iid] = cart.get(iid, 0) + 1
    save_cart(cart)
    return jsonify({"success": True, "count": sum(cart.values())})

@app.route("/api/cart/remove", methods=["POST"])
def api_cart_remove():
    data = request.get_json()
    iid = data.get("item_id")
    cart = get_cart()
    if iid in cart:
        cart[iid] -= 1
        if cart[iid] <= 0: del cart[iid]
    save_cart(cart)
    return jsonify({"success": True, "count": sum(cart.values())})

@app.route("/api/cart/clear", methods=["POST"])
def api_cart_clear():
    save_cart({})
    return jsonify({"success": True})

@app.route("/api/addresses", methods=["GET"])
def api_get_addresses():
    return jsonify(session.get("saved_addresses", []))

@app.route("/api/addresses/save", methods=["POST"])
def api_save_address():
    data = request.get_json() or {}
    addresses = session.get("saved_addresses", [])
    # avoid duplicate by label
    addresses = [a for a in addresses if a.get("label") != data.get("label")]
    addresses.insert(0, data)
    addresses = addresses[:5]  # keep max 5
    session["saved_addresses"] = addresses
    return jsonify({"success": True, "addresses": addresses})

@app.route("/api/addresses/delete", methods=["POST"])
def api_delete_address():
    data = request.get_json() or {}
    addresses = session.get("saved_addresses", [])
    addresses = [a for a in addresses if a.get("label") != data.get("label")]
    session["saved_addresses"] = addresses
    return jsonify({"success": True})

@app.route("/api/order/place", methods=["POST"])
def api_place_order():
    cart = get_cart()
    if not cart: return jsonify({"error": "Cart is empty"}), 400
    data = request.get_json() or {}
    items, total = [], 0
    for iid, qty in cart.items():
        item = get_item(iid)
        if item:
            sub = item["price"] * qty
            total += sub
            items.append({**item, "quantity": qty, "subtotal": sub})
    order_id = "SR" + str(uuid.uuid4())[:8].upper()
    ORDERS[order_id] = {
        "order_id": order_id, "items": items, "total": total,
        "name": data.get("name", "Guest"), "address": data.get("address", ""),
        "status": "confirmed", "placed_at": time.time()
    }
    save_cart({})
    return jsonify({"success": True, "order_id": order_id, "total": total})

function spiceMeter(level) {
  if (level === 0) return `<div class="spice-meter"><span class="spice-label">No Spice</span><span style="font-size:13px;color:var(--muted)">🧊 Mild</span></div>`;
  const labels = ['','Mild','Medium','Spicy','Very Spicy','Fiery Hot!'];
  const colors = ['','#FFA726','#FF7043','#E53935','#B71C1C','#7B1FA2'];
  let flames = '';
  for (let i=1; i<=5; i++) {
    flames += `<span class="spice-flame ${i<=level?'lit':'dim'}" style="${i<=level?'filter:sepia(100%) saturate(3) hue-rotate(-15deg)':''}">🔥</span>`;
  }
  return `<div class="spice-meter"><span class="spice-label" style="color:${colors[level]}">${labels[level]}</span><div class="spice-flames">${flames}</div></div>`;
}

  if(!area){toast('⚠️ Enter area');return;}
  if(!city){toast('⚠️ Select city');return;}

const MAP_WAYPOINTS = [

  {left:72, top:60},  // home

function animateMapRider() {
  const rider = document.getElementById('mapRider');
  if (!rider) return;
  rider.style.left = MAP_WAYPOINTS[0].left + '%';
  rider.style.top  = MAP_WAYPOINTS[0].top  + '%';
  const totalMs = 36000;
  const segMs = totalMs / (MAP_WAYPOINTS.length - 1);
  MAP_WAYPOINTS.forEach((wp, i) => {
    if (i === 0) return;
    setTimeout(()=>{
      rider.style.left = wp.left + '%';
      rider.style.top  = wp.top  + '%';
    }, i * segMs);
  });
}

function initScratch() {

  document.getElementById('srEmoji').textContent = scratchPrize.emoji;

  grad.addColorStop(1,'#7B1FA2');
  scratchCtx.fillStyle = grad;
  scratchCtx.fillRect(0,0,240,140);
  scratchCtx.fillStyle='rgba(255,255,255,0.1)';

  scratchCtx.font='12px Outfit,sans-serif';
  scratchCtx.fillStyle='rgba(255,255,255,0.8)';
  scratchCtx.fillText('Use your finger or mouse',120,90);
  scratchCtx.globalCompositeOperation='destination-out';
  canvas.addEventListener('mousedown',()=>scratchDrawing=true);
  canvas.addEventListener('mouseup',()=>{ scratchDrawing=false; checkScratch(); });
  canvas.addEventListener('mousemove',e=>{ if(!scratchDrawing) return; scratchDraw(e.offsetX,e.offsetY); });
  canvas.addEventListener('touchstart',e=>{ scratchDrawing=true; e.preventDefault(); });
  canvas.addEventListener('touchend',()=>{ scratchDrawing=false; checkScratch(); });
  canvas.addEventListener('touchmove',e=>{
    if(!scratchDrawing) return;
    const r=canvas.getBoundingClientRect();
    scratchDraw(e.touches[0].clientX-r.left,e.touches[0].clientY-r.top);
    e.preventDefault();
  },{passive:false});
}

function scratchDraw(x,y){
  scratchCtx.beginPath();
  scratchCtx.arc(x,y,22,0,Math.PI*2);
  scratchCtx.fill();
}

function checkScratch(){
  const d=scratchCtx.getImageData(0,0,240,140).data;
  let erased=0;
  for(let i=3;i<d.length;i+=4) if(d[i]<128) erased++;
  const pct=erased/(240*140)*100;
  if(pct>40){
    const canvas=document.getElementById('scratchCanvas');
    canvas.style.transition='opacity .5s';
    canvas.style.opacity='0';
    setTimeout(()=>{ canvas.style.display='none'; toast('🎉 Code revealed: '+scratchPrize.code); },500);
  }
}

function resetScratch(){
  const canvas=document.getElementById('scratchCanvas');
  canvas.style.display='block';
  canvas.style.opacity='1';
  canvas.getContext('2d').clearRect(0,0,240,140);
  // Re-init
  canvas.replaceWith(canvas.cloneNode());
  // re-query
  const newCanvas=document.getElementById('scratchCanvas');
  newCanvas.width=240; newCanvas.height=140;
  initScratch();
}
